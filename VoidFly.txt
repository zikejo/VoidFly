local function createButton(data)
    local btn = Instance.new("TextButton")
    btn.Name = data.name
    btn.Parent = frame
    btn.BackgroundColor3 = Color3.fromRGB(25, 10, 55)
    btn.BackgroundTransparency = 0.2
    btn.BorderSizePixel = 0
    btn.Font = Enum.Font.GothamBold
    btn.Text = data.text
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 13
    btn.AutoButtonColor = false

    local w = data.w or 1
    local gap = 6
    local btnW = 34
    local btnH = 28
    local x = (data.col - 1) * (btnW + gap) + 8
    local y = 42 + (data.row - 1) * (btnH + gap)
    btn.Size = UDim2.new(0, btnW * w + (w-1) * gap, 0, btnH)
    btn.Position = UDim2.new(0, x, 0, y)

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = btn

    local s = Instance.new("UIStroke")
    s.Parent = btn
    s.Color = Color3.fromRGB(80, 40, 160)
    s.Thickness = 1.2

    -- Ховер-эффект
    btn.MouseEnter:Connect(function()
        btn.BackgroundColor3 = Color3.fromRGB(50, 20, 110)
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        s.Color = Color3.fromRGB(160, 100, 255)
    end)
    btn.MouseLeave:Connect(function()
        btn.BackgroundColor3 = Color3.fromRGB(25, 10, 55)
        btn.TextColor3 = Color3.fromRGB(210, 190, 255)
        s.Color = Color3.fromRGB(80, 40, 160)
    end)

    return btn
end

-- ======== FLIGHT LOGIC ========
local flying = false
local noclipEnabled = false
local speeds = 1
local bodyVelocity = nil
local bodyGyro = nil
local heartbeat = nil
local noclipConns = {}

function startFly()
    if flying then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")
    if not root or not hum then return end
    flying = true
    hum.PlatformStand = true

    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    bodyVelocity.Parent = root

    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    bodyGyro.Parent = root
    bodyGyro.CFrame = root.CFrame

    local UIS = game:GetService("UserInputService")
    local cam = workspace.CurrentCamera

    heartbeat = game:GetService("RunService").Heartbeat:Connect(function()
        if not flying or not root or not root.Parent then return end
        local f = cam.CFrame.LookVector
        local r = cam.CFrame.RightVector
        local u = cam.CFrame.UpVector
        local move = Vector3.new()
        if UIS:IsKeyDown(Enum.KeyCode.W) then move = move + f end
        if UIS:IsKeyDown(Enum.KeyCode.S) then move = move - f end
        if UIS:IsKeyDown(Enum.KeyCode.A) then move = move - r end
        if UIS:IsKeyDown(Enum.KeyCode.D) then move = move + r end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then move = move + u end
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - u end
        if move.Magnitude > 0 then move = move.Unit * (speeds * 20) end
        bodyVelocity.Velocity = move
        bodyGyro.CFrame = CFrame.lookAt(root.Position, root.Position + f)
    end)
end

function stopFly()
    flying = false
    local char = player.Character
    if char then
        local hum = char:FindFirstChild("Humanoid")
        if hum then hum.PlatformStand = false end
    end
    if bodyVelocity then bodyVelocity:Destroy(); bodyVelocity = nil end
    if bodyGyro then bodyGyro:Destroy(); bodyGyro = nil end
    if heartbeat then heartbeat:Disconnect(); heartbeat = nil end
end

function toggleNoclip()
    noclipEnabled = not noclipEnabled
    if noclipEnabled then
        btnNoclip.Text = "NOCLIP: ON"
        btnNoclip.BackgroundColor3 = Color3.fromRGB(60, 0, 100)
        for _, conn in pairs(noclipConns) do conn:Disconnect() end
        noclipConns = {}
        local conn = game:GetService("RunService").Stepped:Connect(function()
            if not noclipEnabled then return end
            local char = player.Character
            if not char then return end
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end)
        table.insert(noclipConns, conn)
    else
        btnNoclip.Text = "NOCLIP: OFF"
        btnNoclip.BackgroundColor3 = Color3.fromRGB(25, 10, 55)
        for _, conn in pairs(noclipConns) do conn:Disconnect() end
        noclipConns = {}
        local char = player.Character
        if char then
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = true
                end
            end
        end
    end
end

        local char = player.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame + workspace.CurrentCamera.CFrame.UpVector * 3
        end
    end
end)

btnDown.MouseButton1Down:Connect(function()
    if flying then
        local char = player.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame - workspace.CurrentCamera.CFrame.UpVector * 3
        end
    end
end)

btnPlus.MouseButton1Down:Connect(function()
    if speeds < 10 then
        speeds = speeds + 1
        btnSpeed.Text = speeds
    end
end)

btnMinus.MouseButton1Down:Connect(function()
    if speeds > 1 then
        speeds = speeds - 1
        btnSpeed.Text = speeds
    end
end)

player.CharacterAdded:Connect(function()
    if flying then stopFly(); btnFly.Text = "✦ FLY" end
    if noclipEnabled then
        for _, conn in pairs(noclipConns) do conn:Disconnect() end
        noclipConns = {}
        noclipEnabled = false
        btnNoclip.Text = "NOCLIP: OFF"
        btnNoclip.BackgroundColor3 = Color3.fromRGB(25, 10, 55)
    end
end)
